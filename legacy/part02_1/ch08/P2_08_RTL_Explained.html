개념

1. AMD 제공 XADC Wizard를 재사용하는 이유

7시리즈 XADC는 FPGA 내부의 아날로그-디지털 변환 하드 매크로입니다. 08장에서는 xadc_wiz_0을 AXI4-Lite로 설정해 MicroBlaze V가 결과를 읽습니다. 새로운 12비트 ADC RTL을 자체 구현하지 않습니다.

Top에는 always·if·ADC 계산기·기능 FSM이 없습니다. 07장의 LED0/1/2·FND·모터 출력 경로는 보존하고, XADC 아날로그 포트만 연결합니다.

재사용

07장 motor_controller

`timescale 1ns/1ps
// P2-07 reusable motor controller; no AXI or board-specific pins.
// (추가) Valid, recent HC-SR04 data + AUTO must exist before any drive.
// (추가) Near/fault/timeout immediately disables ENA via combinational gating.
// (추가) Fault while commanded latches TRIP; MOTOR STOP clears, FWD/REV re-arms.
module motor_controller #(
    parameter integer DEADTIME_CYCLES = 10000000, // 100 ms at 100MHz
    parameter integer FRESH_CYCLES    = 25000000  // 250 ms at 100MHz
)(
    input wire clk, input wire resetn,
    input wire [1:0] request_i, // 0 STOP, 1 FWD, 2 REV
    input wire stop_clear_i,   // One-clock pulse on MOTOR STOP write
    input wire sonic_auto_i, input wire sonic_done_i,
    input wire sonic_valid_i, input wire sonic_unsafe_i,
    output wire motor_in1_o, output wire motor_in2_o, output wire motor_ena_o,
    output wire [1:0] active_dir_o,
    output wire sensor_safe_o, output wire fresh_o,
    output reg trip_latched_o, output wire deadtime_o
);
    reg [31:0] fresh_count, dead_count;
    reg [1:0] drive_dir;
    // Always gate the physical outputs, even before the next clock edge.
    assign fresh_o = (fresh_count != 32'd0);
    assign sensor_safe_o = sonic_auto_i && fresh_o &&
                           sonic_valid_i && !sonic_unsafe_i;
    wire output_allowed = sensor_safe_o && !trip_latched_o &&
                          !stop_clear_i && (request_i != 2'd0);
    assign motor_in1_o = output_allowed && (drive_dir == 2'd1);
    assign motor_in2_o = output_allowed && (drive_dir == 2'd2);
    assign motor_ena_o = motor_in1_o || motor_in2_o;
    assign active_dir_o = motor_ena_o ? drive_dir : 2'd0;
    assign deadtime_o = (dead_count != 32'd0);

    always @(posedge clk) begin
        if (!resetn) begin
            fresh_count <= 32'd0;
            dead_count <= 32'd0;
            drive_dir <= 2'd0;
            trip_latched_o <= 1'b0;
        end else begin
            // Stale/invalid/off -> stop. Periodic successful DONE re-arms freshness.
            if (!sonic_auto_i || !sonic_valid_i || sonic_unsafe_i)
                fresh_count <= 32'd0;
            else if (sonic_done_i)
                fresh_count <= FRESH_CYCLES;
            else if (fresh_count != 32'd0)
                fresh_count <= fresh_count - 32'd1;

            if (stop_clear_i)
                trip_latched_o <= 1'b0;
            else if (request_i != 2'd0 && !sensor_safe_o)
                trip_latched_o <= 1'b1;

            if (!sensor_safe_o || trip_latched_o || stop_clear_i ||
                request_i == 2'd0 || request_i == 2'd3) begin
                drive_dir <= 2'd0;
                // (변경) STOP after motion must not bypass reversal dead-time.
                if (drive_dir != 2'd0)
                    dead_count <= DEADTIME_CYCLES;
                else if (dead_count != 32'd0)
                    dead_count <= dead_count - 32'd1;
            end else if (drive_dir != 2'd0 && drive_dir != request_i) begin
                // Disable bridge, then enforce a direction-change dead time.
                drive_dir <= 2'd0;
                dead_count <= DEADTIME_CYCLES;
            end else if (drive_dir == 2'd0) begin
                if (dead_count != 32'd0)
                    dead_count <= dead_count - 32'd1;
                else
                    drive_dir <= request_i;
            end
        end
    end
endmodule

07장 motor AXI slave

`timescale 1ns/1ps
// P2-07 AXI4-Lite 32-bit motor register block, separate AW/W handshakes.
// Register map: 0x00 CMD RW 0 STOP/1 FWD/2 REV;
// 0x04 STATUS RO; 0x08 ID 0x00070001;
// 0x0C DEADTIME_CYCLES RO; 0x10 FRESH_CYCLES RO; 0x14 CLK_HZ RO.
module p2_07_axi_motor_v1_0_S00_AXI #(
    parameter integer C_S_AXI_DATA_WIDTH=32,
    parameter integer C_S_AXI_ADDR_WIDTH=6,
    parameter integer CLK_HZ=100000000,
    parameter integer DEADTIME_CYCLES=10000000,
    parameter integer FRESH_CYCLES=25000000
)(
    input wire S_AXI_ACLK, input wire S_AXI_ARESETN,
    input wire [C_S_AXI_ADDR_WIDTH-1:0] S_AXI_AWADDR,
    input wire [2:0] S_AXI_AWPROT, input wire S_AXI_AWVALID,
    output wire S_AXI_AWREADY,
    input wire [C_S_AXI_DATA_WIDTH-1:0] S_AXI_WDATA,
    input wire [(C_S_AXI_DATA_WIDTH/8)-1:0] S_AXI_WSTRB,
    input wire S_AXI_WVALID, output wire S_AXI_WREADY,
    output reg [1:0] S_AXI_BRESP, output reg S_AXI_BVALID,
    input wire S_AXI_BREADY,
    input wire [C_S_AXI_ADDR_WIDTH-1:0] S_AXI_ARADDR,
    input wire [2:0] S_AXI_ARPROT, input wire S_AXI_ARVALID,
    output wire S_AXI_ARREADY,
    output reg [C_S_AXI_DATA_WIDTH-1:0] S_AXI_RDATA,
    output reg [1:0] S_AXI_RRESP, output reg S_AXI_RVALID,
    input wire S_AXI_RREADY,
    input wire sonic_auto_i, input wire sonic_done_i,
    input wire sonic_valid_i, input wire sonic_unsafe_i,
    output wire motor_in1_o, output wire motor_in2_o, output wire motor_ena_o
);
    localparam [1:0] OKAY=2'b00, SLVERR=2'b10;
    reg [1:0] requested_dir;
    reg stop_clear;
    wire [1:0] active_dir;
    wire sensor_safe, fresh, trip, deadtime;
    motor_controller #(.DEADTIME_CYCLES(DEADTIME_CYCLES),
                       .FRESH_CYCLES(FRESH_CYCLES)) u_motor (
        .clk(S_AXI_ACLK),.resetn(S_AXI_ARESETN),
        .request_i(requested_dir),.stop_clear_i(stop_clear),
        .sonic_auto_i(sonic_auto_i),.sonic_done_i(sonic_done_i),
        .sonic_valid_i(sonic_valid_i),.sonic_unsafe_i(sonic_unsafe_i),
        .motor_in1_o(motor_in1_o),.motor_in2_o(motor_in2_o),
        .motor_ena_o(motor_ena_o),.active_dir_o(active_dir),
        .sensor_safe_o(sensor_safe),.fresh_o(fresh),
        .trip_latched_o(trip),.deadtime_o(deadtime)
    );
    wire blocked = (requested_dir != 2'd0) && (!sensor_safe || trip);
    wire [31:0] status_reg = {22'd0, blocked, (requested_dir != 2'd0),
                             sonic_auto_i, fresh, sensor_safe, deadtime,
                             trip, (active_dir == 2'd2),
                             (active_dir == 2'd1), motor_ena_o};
    reg aw_hold, w_hold;
    reg [C_S_AXI_ADDR_WIDTH-1:0] awaddr_hold;
    reg [31:0] wdata_hold;
    reg [3:0] wstrb_hold;
    reg [31:0] merged;
    integer j;
    assign S_AXI_AWREADY = S_AXI_ARESETN && !aw_hold && !S_AXI_BVALID;
    assign S_AXI_WREADY  = S_AXI_ARESETN && !w_hold && !S_AXI_BVALID;
    assign S_AXI_ARREADY = S_AXI_ARESETN && !S_AXI_RVALID;

    always @(posedge S_AXI_ACLK) begin
        if (!S_AXI_ARESETN) begin
            requested_dir <= 2'd0;
            stop_clear <= 1'b0;
            aw_hold <= 1'b0; w_hold <= 1'b0;
            awaddr_hold <= {C_S_AXI_ADDR_WIDTH{1'b0}};
            wdata_hold <= 32'd0; wstrb_hold <= 4'd0;
            S_AXI_BRESP <= OKAY;
            S_AXI_BVALID <= 1'b0;
        end else begin
            stop_clear <= 1'b0;
            if (S_AXI_AWVALID && S_AXI_AWREADY) begin
                aw_hold <= 1'b1;
                awaddr_hold <= S_AXI_AWADDR;
            end
            if (S_AXI_WVALID && S_AXI_WREADY) begin
                w_hold <= 1'b1;
                wdata_hold <= S_AXI_WDATA;
                wstrb_hold <= S_AXI_WSTRB;
            end
            if (S_AXI_BVALID && S_AXI_BREADY) S_AXI_BVALID <= 1'b0;
            if (aw_hold && w_hold && !S_AXI_BVALID) begin
                aw_hold <= 1'b0; w_hold <= 1'b0;
                S_AXI_BVALID <= 1'b1;
                S_AXI_BRESP <= OKAY;
                if (awaddr_hold == 6'h00) begin
                    merged = {30'd0,requested_dir};
                    for (j=0;j<4;j=j+1)
                        if (wstrb_hold[j])
                            merged[j*8 +: 8] = wdata_hold[j*8 +: 8];
                    if (merged > 32'd2) begin
                        S_AXI_BRESP <= SLVERR;
                    end else if (wstrb_hold != 4'b0000) begin
                        requested_dir <= merged[1:0];
                        if (merged == 32'd0) stop_clear <= 1'b1;
                    end
                end else begin
                    S_AXI_BRESP <= SLVERR;
                end
            end
        end
    end
    always @(posedge S_AXI_ACLK) begin
        if (!S_AXI_ARESETN) begin
            S_AXI_RVALID <= 1'b0;
            S_AXI_RDATA <= 32'd0;
            S_AXI_RRESP <= OKAY;
        end else begin
            if (S_AXI_RVALID && S_AXI_RREADY) S_AXI_RVALID <= 1'b0;
            if (S_AXI_ARVALID && S_AXI_ARREADY) begin
                S_AXI_RVALID <= 1'b1;
                S_AXI_RRESP <= OKAY;
                case (S_AXI_ARADDR)
                6'h00: S_AXI_RDATA <= {30'd0,requested_dir};
                6'h04: S_AXI_RDATA <= status_reg;
                6'h08: S_AXI_RDATA <= 32'h00070001;
                6'h0C: S_AXI_RDATA <= DEADTIME_CYCLES;
                6'h10: S_AXI_RDATA <= FRESH_CYCLES;
                6'h14: S_AXI_RDATA <= CLK_HZ;
                default: begin
                    S_AXI_RDATA <= 32'd0;
                    S_AXI_RRESP <= SLVERR;
                end
                endcase
            end
        end
    end
endmodule

06장 HC-SR04 코어

`timescale 1ns/1ps
// P2-06 HC-SR04 reusable timing core; no board pins or AXI dependency.
// (추가) 2FF Echo synchronizer, microsecond tick, bounded waits, cooldown.
// 100 MHz AXI clock -> 100 clocks / 1 us; 12 us TRIG, >=60 ms cooldown.
// Result is the last COMPLETED reading. valid_o and fault flags persist.
module hc_sr04_core #(
    parameter integer CLK_HZ = 100000000,
    parameter integer TRIG_US = 12,
    parameter integer TIMEOUT_US = 38000,
    parameter integer COOLDOWN_US = 60000
)(
    input wire clk, input wire resetn,
    input wire echo_i, input wire start_i,
    output reg trig_o, output wire ready_o, output wire busy_o,
    output reg done_pulse_o,
    output reg [15:0] distance_cm_o,
    output reg [31:0] echo_us_o,
    output reg valid_o, output reg timeout_o, output reg out_range_o
);
    localparam integer TICK_CYCLES = CLK_HZ / 1000000;
    localparam [2:0] S_IDLE=3'd0, S_TRIG=3'd1,
                     S_WAIT=3'd2, S_ECHO=3'd3, S_COOL=3'd4;
    reg [2:0] state;
    reg [31:0] div_count, us_count, high_us;
    reg echo_meta, echo_sync, echo_prev;
    wire tick_us = (div_count == TICK_CYCLES-1);
    wire echo_rise = echo_sync && !echo_prev;
    reg echo_rise_pending; // Catch sub-us rising edges until the next us tick.
    assign ready_o = (state == S_IDLE);
    assign busy_o = !ready_o;

    // (추가) FPGA input can be asynchronous. Metastability cannot be
    // eliminated absolutely; this 2FF path lowers the risk.
    always @(posedge clk) begin
        if (!resetn) begin
            echo_meta <= 1'b0;
            echo_sync <= 1'b0;
            echo_prev <= 1'b0;
        end else begin
            echo_meta <= echo_i;
            echo_sync <= echo_meta;
            echo_prev <= echo_sync;
        end
    end

    always @(posedge clk) begin
        if (!resetn || state != S_WAIT) echo_rise_pending <= 1'b0;
        else if (echo_rise) echo_rise_pending <= 1'b1;
        else if (tick_us) echo_rise_pending <= 1'b0;
    end

    always @(posedge clk) begin
        if (!resetn) begin
            div_count <= 32'd0;
            state <= S_IDLE;
            us_count <= 32'd0;
            high_us <= 32'd0;
            trig_o <= 1'b0;
            done_pulse_o <= 1'b0;
            distance_cm_o <= 16'd0;
            echo_us_o <= 32'd0;
            valid_o <= 1'b0;
            timeout_o <= 1'b0;
            out_range_o <= 1'b0;
        end else begin
            done_pulse_o <= 1'b0;
            if (tick_us) div_count <= 32'd0;
            else div_count <= div_count + 32'd1;

            case (state)
            S_IDLE: begin
                if (start_i) begin
                    trig_o <= 1'b1;
                    us_count <= 32'd0;
                    high_us <= 32'd0;
                    state <= S_TRIG;
                end
            end
            S_TRIG: if (tick_us) begin
                if (us_count + 32'd1 >= TRIG_US) begin
                    trig_o <= 1'b0;
                    us_count <= 32'd0;
                    state <= S_WAIT;
                end else us_count <= us_count + 32'd1;
            end
            S_WAIT: if (tick_us) begin
                if (echo_rise_pending || echo_rise) begin
                    // Edge latched at 100MHz to avoid missing it on us tick.
                    high_us <= 32'd1;
                    us_count <= 32'd0;
                    state <= S_ECHO;
                end else if (us_count + 32'd1 >= TIMEOUT_US) begin
                    valid_o <= 1'b0;
                    timeout_o <= 1'b1;
                    out_range_o <= 1'b0;
                    distance_cm_o <= 16'd0;
                    echo_us_o <= 32'd0;
                    done_pulse_o <= 1'b1;
                    us_count <= 32'd0;
                    state <= S_COOL;
                end else us_count <= us_count + 32'd1;
            end
            S_ECHO: if (tick_us) begin
                if (!echo_sync) begin
                    echo_us_o <= high_us;
                    // D(cm) ~= EchoHigh(us)/58 at nominal room temperature.
                    if (high_us >= 32'd116 && high_us <= 32'd23200) begin
                        distance_cm_o <= (high_us + 32'd29) / 32'd58;
                        valid_o <= 1'b1;
                        timeout_o <= 1'b0;
                        out_range_o <= 1'b0;
                    end else begin
                        distance_cm_o <= 16'd0;
                        valid_o <= 1'b0;
                        timeout_o <= 1'b0;
                        out_range_o <= 1'b1;
                    end
                    done_pulse_o <= 1'b1;
                    us_count <= 32'd0;
                    state <= S_COOL;
                end else if (high_us >= TIMEOUT_US) begin
                    echo_us_o <= high_us;
                    distance_cm_o <= 16'd0;
                    valid_o <= 1'b0;
                    timeout_o <= 1'b1;
                    out_range_o <= 1'b0;
                    done_pulse_o <= 1'b1;
                    us_count <= 32'd0;
                    state <= S_COOL;
                end else high_us <= high_us + 32'd1;
            end
            S_COOL: if (tick_us) begin
                if (us_count + 32'd1 >= COOLDOWN_US) begin
                    us_count <= 32'd0;
                    state <= S_IDLE;
                end else us_count <= us_count + 32'd1;
            end
            default: state <= S_IDLE;
            endcase
        end
    end
endmodule

04장 PWM 코어

`timescale 1ns/1ps
// P2-04: 보드와 무관한 단일 채널 범용 PWM 코어.
// (추가) period와 duty는 클록 횟수 단위. 0%/100%는 지속 LOW/HIGH.
// (추가) 설정은 주기 마지막 경계에서만 활성값으로 반영.
module pwm_core #(
    parameter [31:0] RESET_PERIOD = 32'd100000
)(
    input  wire        clk,
    input  wire        resetn,
    input  wire        cfg_apply,
    input  wire [31:0] cfg_period,
    input  wire [31:0] cfg_duty,
    input  wire        cfg_enable,
    input  wire        cfg_polarity,
    output wire        pwm_out,
    output reg         pending,
    output reg [31:0] active_period,
    output reg [31:0] active_duty,
    output reg         active_enable,
    output reg         active_polarity
);
    reg [31:0] counter;
    reg [31:0] shadow_period;
    reg [31:0] shadow_duty;
    reg        shadow_enable;
    reg        shadow_polarity;

    wire boundary = (counter >= active_period - 32'd1);
    wire duty_high = (counter < active_duty);

    // (주의) enable=0이면 polarity와 관계없이 LOW로 차단.
    // (설명) 기본 polarity에서 active_duty=0이면 항상 LOW,
    //         active_duty=active_period이면 항상 HIGH.
    assign pwm_out = resetn && active_enable &&
                     (active_polarity ? !duty_high : duty_high);

    always @(posedge clk) begin
        if (!resetn) begin
            counter         <= 32'd0;
            active_period   <= RESET_PERIOD;
            active_duty     <= 32'd0;
            active_enable   <= 1'b0;
            active_polarity <= 1'b0;
            shadow_period   <= RESET_PERIOD;
            shadow_duty     <= 32'd0;
            shadow_enable   <= 1'b0;
            shadow_polarity <= 1'b0;
            pending         <= 1'b0;
        end else begin
            if (boundary) begin
                counter <= 32'd0;
                if (pending) begin
                    active_period   <= shadow_period;
                    active_duty     <= shadow_duty;
                    active_enable   <= shadow_enable;
                    active_polarity <= shadow_polarity;
                    pending         <= 1'b0;
                end
            end else begin
                counter <= counter + 32'd1;
            end
            // (추가) 경계와 APPLY가 겹치면 새 값은 다음 경계에 적용.
            // 여러 APPLY가 겹치면 최종 요청값을 우선하는 단일 Shadow 구조.
            if (cfg_apply) begin
                shadow_period   <= cfg_period;
                shadow_duty     <= cfg_duty;
                shadow_enable   <= cfg_enable;
                shadow_polarity <= cfg_polarity;
                pending         <= 1'b1;
            end
        end
    end
endmodule

6. 설계·검증 경계

AXI XADC 변환값은 CPU가 전압 조회 또는 LED1 PWM 설정으로 사용하지만, 기존 모터 motor_controller의 sonic_unsafe·Deadtime·Trip 해제는 침범하지 않습니다.

08장 추가는 물리 아날로그 핀 두 개와 XADC IP입니다. 기존 PWM IP, CdS 게이트, 모터 제어 RTL의 기능 변경은 없습니다.

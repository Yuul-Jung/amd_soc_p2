01과제 개요

HC-SR501 PIR(Passive Infrared, 피아이알, 수동형 적외선) 센서 모듈은 주변 적외선 변화로 움직임을 감지하면 디지털 출력 OUT을 HIGH로 만든다. FPGA는 이 신호를 단순 GPIO 상태로 읽는 것에서 나아가 비동기 입력 동기화 → 상승 이벤트 검출 → CPU 인터럽트 발생 → ACK 해제로 확장한다.

기본 과제는 센서 초기 안정화, 입력 안정화 필터, 이벤트 횟수 및 인터럽트 Pending 등록이다. 응용 단계에서는 신규 AXI4-Lite 슬레이브 IP를 통해 ARM, IRQ_ENABLE, STATUS, EVENT_COUNT를 제어한다.

02전체 계층 구조

각 RTL 명칭은 실제 모듈과 일치한다. 이전 단계의 GPIO/AXI IP를 덮어쓰지 않고, 독립 PIR IRQ IP를 추가한다. 사용자 Top은 보드 핀·Wire·IP 연결만 담당한다.

04PIR 입력 동기화와 이벤트 처리

실습 명령 · Bash / 실행 예시

    ↓ 2-stage FF Synchronizer

    ↓ FILTER_CYCLES 동안 새 입력 유지

    ↓ 0 → 1 전환 + ARM + WARMUP_READY

    ├── event_count 증가 (포화)
    └── event_pending=1 (sticky)

        CMD.ACK_EVENT=1
          ↓ event_pending=0 → irq=0

센서 출력이 10초 동안 계속 HIGH라고 해도 이 설계에서는 한 번의 상승 전환을 1회로 기록한다. 반대로 센서가 LOW로 돌아왔다가 다시 HIGH가 되면 새로운 이벤트로 등록할 수 있다. event_count는 사람 수를 측정하는 값이 아니다.

ACK를 쓰는 바로 그 클록에 새로운 PIR 상승 에지가 들어오면 새로운 이벤트가 우선하며 Pending이 유지된다. CPU가 IRQ를 껐다 켜도 Pending은 그대로 남아 있을 수 있다.

05IP 제어용 AXI Register Map

OffsetRegisterR/W설명
0x00CTRLR/Wbit0 ARM, bit1 IRQ_ENABLE, 리셋 후 0
0x04CMDWrite onlybit0 ACK_EVENT, bit1 CLEAR_COUNT — W1P
0x08STATUSRead onlybit0 READY, bit1 PIR_LEVEL, bit2 PENDING, bit3 IRQ
0x0CEVENT_COUNTRead only0에서 시작하는 32비트 누적 상승 이벤트 수
0x10IDRead only0x00180001

W1P(Write One Pulse, 1을 쓸 때 1회 명령 발생)는 CMD의 비트를 1로 쓰면 한 클록 펄스가 발생하고 자동으로 해제되는 방식이다. WSTRB(쓰기 바이트 유효 비트)를 적용하고, 주소/데이터가 서로 다른 클록에 도착하는 AXI AW/W 채널도 처리한다. 금지·비정렬 주소는 SLVERR 응답을 반환한다.

실제 BaseAddr는 아직 정하지 않는다. 향후 Vivado Address Editor에서 기존 GPIO·UART·타이머·모터 IP와 중복되지 않는 주소를 할당하고 XSA/BSP를 확인한다.

CPU 인터럽트 처리 순서는 인터페이스 계약이며, 실제 Vitis BSP·인터럽트 핸들러·AXI Interrupt Controller 연결은 후속 실습에서 확인한다.

07전체 RTL과 자가확인 테스트벤치

기본 Controller와 AXI Wrapper, 두 개의 독립 테스트벤치를 한 곳에 모았다. 기존 성공 Verilog RTL은 수정하지 않는다.

명령·출력
`timescale 1ns/1ps
// P2-18 HC-SR501 PIR event controller (board-independent).
// pir_in_async -> 2-FF synchronizer -> stable filter -> rising event.
// Warm-up masks initial module transients. Pending is sticky until ACK;
// if ACK and new event coincide, NEW EVENT WINS (no lost interrupt).
// IRQ is level-sensitive: irq_enable & event_pending.
// This RTL does NOT implement a physical AXI-Lite bus / CPU interrupt controller.
module pir_irq_controller #(
    parameter [63:0] WARMUP_CYCLES = 64'd6000000000, // 60 sec @100 MHz
    parameter integer FILTER_CYCLES = 3
)(
    input wire clk,
    input wire resetn,
    input wire pir_in_async,
    input wire arm,
    input wire irq_enable,
    input wire irq_ack_pulse,
    input wire count_clear_pulse,
    output wire irq,
    output reg warmup_ready,
    output reg pir_level,
    output reg motion_pulse,
    output reg event_pending,
    output reg [31:0] event_count
);
    // Wide enough for 60s * 100MHz, no 32-bit overflow.
    reg [63:0] warmup_count;
    reg [31:0] filter_count;
    reg pir_meta, pir_sync;
    wire change_pending = (pir_sync != pir_level);
    wire qualified_rise = warmup_ready && arm && !pir_level &&
        pir_sync && change_pending && (filter_count >= FILTER_CYCLES-1);
    assign irq = irq_enable && event_pending;

    always @(posedge clk) begin
        if(!resetn) begin
            pir_meta<=0; pir_sync<=0; pir_level<=0;
            filter_count<=0; warmup_count<=0; warmup_ready<=0;
            motion_pulse<=0; event_pending<=0; event_count<=0;
        end else begin
            // (추가) Async PIR input synchronizer.
            pir_meta<=pir_in_async;
            pir_sync<=pir_meta;
            if(!warmup_ready) begin
                if(warmup_count >= WARMUP_CYCLES-1)
                    warmup_ready<=1;
                else warmup_count<=warmup_count+1;
            end

            // (추가) Reject glitches shorter than FILTER_CYCLES.
            if(!change_pending) filter_count<=0;
            else if(filter_count >= FILTER_CYCLES-1) begin
                pir_level<=pir_sync;
                filter_count<=0;
            end else filter_count<=filter_count+1;

            // Sticky event; simultaneous ACK cannot erase a new edge.
            motion_pulse<=qualified_rise;
            if(qualified_rise)
                event_pending<=1;
            else if(irq_ack_pulse)
                event_pending<=0;

            // Saturating number of qualified rising edges. This is NOT
            // a direct count of humans or motion events behind one long
            // HC-SR501 HIGH interval.
            if(count_clear_pulse) event_count<=qualified_rise ? 1 : 0;
            else if(qualified_rise && event_count!=32'hffffffff)
                event_count<=event_count+1;
        end
    end
endmodule

`timescale 1ns/1ps
// P2-18: 32-bit AXI4-Lite register block around pir_irq_controller.
// Register offsets:
// 0x00 CTRL R/W: bit0 ARM, bit1 IRQ_ENABLE
// 0x04 CMD  W: bit0 ACK pending, bit1 CLEAR event_count (W1P)
// 0x08 STATUS R: bit0 warmup_ready, bit1 pir_level,
//                bit2 event_pending, bit3 irq
// 0x0C EVENT_COUNT R
// 0x10 ID R=0x00180001
// AXI WSTRB honored. Illegal/misaligned/unmapped access -> SLVERR.
// GPIO input asynchronous. CPU IRQ is LEVEL (not a one-clock pulse).
module p2_18_axi_pir_irq_v1_0 #(
    parameter [63:0] WARMUP_CYCLES=64'd6000000000,
    parameter integer FILTER_CYCLES=3
)(
    input wire S_AXI_ACLK,input wire S_AXI_ARESETN,
    input wire [5:0] S_AXI_AWADDR,
    input wire [2:0] S_AXI_AWPROT,
    input wire S_AXI_AWVALID,output wire S_AXI_AWREADY,
    input wire [31:0] S_AXI_WDATA,
    input wire [3:0] S_AXI_WSTRB,
    input wire S_AXI_WVALID,output wire S_AXI_WREADY,
    output reg [1:0] S_AXI_BRESP,
    output reg S_AXI_BVALID,input wire S_AXI_BREADY,
    input wire [5:0] S_AXI_ARADDR,
    input wire [2:0] S_AXI_ARPROT,
    input wire S_AXI_ARVALID,output wire S_AXI_ARREADY,
    output reg [31:0] S_AXI_RDATA,
    output reg [1:0] S_AXI_RRESP,
    output reg S_AXI_RVALID,input wire S_AXI_RREADY,
    input wire pir_in_async,output wire irq
);
    localparam [1:0] OKAY=2'b00, SLVERR=2'b10;
    reg arm,irq_enable,ack_strobe,clear_strobe;
    wire ready,level,pulse,pending;
    wire [31:0] count;
    reg aw_hold,w_hold;
    reg [5:0] saved_addr;
    reg [31:0] saved_data;
    reg [3:0] saved_strb;
    pir_irq_controller #(
        .WARMUP_CYCLES(WARMUP_CYCLES),.FILTER_CYCLES(FILTER_CYCLES)
    ) u_core(
        .clk(S_AXI_ACLK),.resetn(S_AXI_ARESETN),
        .pir_in_async(pir_in_async),.arm(arm),
        .irq_enable(irq_enable),
        .irq_ack_pulse(ack_strobe),.count_clear_pulse(clear_strobe),
        .irq(irq),.warmup_ready(ready),
        .pir_level(level),.motion_pulse(pulse),
        .event_pending(pending),.event_count(count)
    );
    assign S_AXI_AWREADY=S_AXI_ARESETN && !aw_hold && !S_AXI_BVALID;
    assign S_AXI_WREADY =S_AXI_ARESETN && !w_hold && !S_AXI_BVALID;
    assign S_AXI_ARREADY=S_AXI_ARESETN && !S_AXI_RVALID;
    always @(posedge S_AXI_ACLK)begin
        if(!S_AXI_ARESETN)begin
            arm<=0;irq_enable<=0;ack_strobe<=0;clear_strobe<=0;
            aw_hold<=0;w_hold<=0;saved_addr<=0;saved_data<=0;
            saved_strb<=0;S_AXI_BRESP<=OKAY;S_AXI_BVALID<=0;
            S_AXI_RRESP<=OKAY;S_AXI_RVALID<=0;S_AXI_RDATA<=0;
        end else begin
            ack_strobe<=0;clear_strobe<=0;
            if(S_AXI_AWVALID && S_AXI_AWREADY)begin
                saved_addr<=S_AXI_AWADDR;aw_hold<=1;
            end
            if(S_AXI_WVALID && S_AXI_WREADY)begin
                saved_data<=S_AXI_WDATA;saved_strb<=S_AXI_WSTRB;
                w_hold<=1;
            end
            if(S_AXI_BVALID && S_AXI_BREADY)S_AXI_BVALID<=0;
            if(aw_hold && w_hold && !S_AXI_BVALID)begin
                aw_hold<=0;w_hold<=0;S_AXI_BVALID<=1;
                S_AXI_BRESP<=OKAY;
                case(saved_addr)
                6'h00:if(saved_strb[0])begin
                    arm<=saved_data[0];irq_enable<=saved_data[1];
                end
                6'h04:if(saved_strb[0])begin
                    ack_strobe<=saved_data[0];
                    clear_strobe<=saved_data[1];
                end
                default:S_AXI_BRESP<=SLVERR;
                endcase
            end
            if(S_AXI_RVALID && S_AXI_RREADY)S_AXI_RVALID<=0;
            if(S_AXI_ARVALID && S_AXI_ARREADY)begin
                S_AXI_RVALID<=1;S_AXI_RRESP<=OKAY;
                case(S_AXI_ARADDR)
                6'h00:S_AXI_RDATA<={30'b0,irq_enable,arm};
                6'h08:S_AXI_RDATA<={28'b0,irq,pending,level,ready};
                6'h0c:S_AXI_RDATA<=count;
                6'h10:S_AXI_RDATA<=32'h00180001;
                default:begin S_AXI_RRESP<=SLVERR;S_AXI_RDATA<=0;end
                endcase
            end
        end
    end
endmodule

`timescale 1ns/1ps
module tb_pir_irq_controller;
    reg clk=0; always #5 clk=~clk;
    reg resetn=0, pir=0, arm=1, enable=1, ack=0, clear=0;
    wire irq,ready,level,pulse,pending;
    wire [31:0] count;
    integer pulses=0;
    pir_irq_controller #(
        .WARMUP_CYCLES(64'd20), .FILTER_CYCLES(3)
    ) dut (
        .clk(clk),.resetn(resetn),.pir_in_async(pir),
        .arm(arm),.irq_enable(enable),
        .irq_ack_pulse(ack),.count_clear_pulse(clear),
        .irq(irq),.warmup_ready(ready),.pir_level(level),
        .motion_pulse(pulse),.event_pending(pending),.event_count(count)
    );
    always @(posedge clk) if (pulse) pulses=pulses+1;
    task advance;
        input integer cycles;
        integer k;
        begin for(k=0;k<cycles;k=k+1) @(negedge clk); end
    endtask
    task set_pir;
        input value;
        begin @(negedge clk);pir=value; end
    endtask
    task do_ack;
        begin @(negedge clk);ack=1;@(negedge clk);ack=0;end
    endtask
    initial begin
        advance(4);resetn=1;
        set_pir(1);advance(9);
        if(ready || count!=0 || pending || irq)
            $fatal(1,"warmup early event");
        advance(20);
        if(!ready || !level || pending || count!=0)
            $fatal(1,"warmup high should be baseline, not event");
        $display("P2_18_PIR_WARMUP_MASK_PASS");

        set_pir(0);advance(9);
        if(level!==0) $fatal(1,"falling edge not synchronized");
        // 1-cycle glitch is shorter than FILTER_CYCLES, ignore.
        set_pir(1);advance(1);set_pir(0);advance(9);
        if(level || count || pending) $fatal(1,"glitch counted");
        $display("P2_18_PIR_GLITCH_FILTER_PASS");

        set_pir(1);advance(11);
        if(!level || !pending || !irq || count!=1 || pulses!=1)
            $fatal(1,"rising event count=%0d pulses=%0d",count,pulses);
        advance(30);
        if(count!=1 || pulses!=1) $fatal(1,"stuck HIGH duplicate event");
        $display("P2_18_PIR_RISE_ONLY_STICKY_IRQ_PASS");

        // Mask IRQ without clearing pending; re-enable reveals pending.
        @(negedge clk);enable=0;
        #1;
        if(irq || !pending) $fatal(1,"enable mask failure");
        @(negedge clk);enable=1;
        #1;
        if(!irq) $fatal(1,"unmask pending failure");
        do_ack();advance(2);
        if(irq || pending || count!=1)
            $fatal(1,"ack failed or counter lost");
        $display("P2_18_PIR_IRQ_ENABLE_ACK_PASS");

        // Returning LOW and then HIGH creates another event.
        set_pir(0);advance(9);
        set_pir(1);advance(10);
        if(count!=2 || !irq || pulses!=2)
            $fatal(1,"second motion not counted");
        do_ack();
        $display("P2_18_PIR_RETRIGGER_PASS");

        // Disable ARM, next high does not create event;
        // enabling ARM while sensor stays high also does not.
        set_pir(0);advance(10);
        @(negedge clk);arm=0;
        set_pir(1);advance(10);
        if(count!=2 || pending || irq)
            $fatal(1,"arm gate failed");
        @(negedge clk);arm=1;advance(7);
        if(count!=2 || pending) $fatal(1,"arm caused fake rise");
        $display("P2_18_PIR_ARM_GATE_PASS");

        set_pir(0);advance(10);
        // ACK coincides with new event: event MUST win.
        set_pir(1);
        wait(dut.qualified_rise);
        @(negedge clk);ack=1;
        @(negedge clk);ack=0;
        advance(4);
        if(count!=3 || !pending || !irq || pulses!=3)
            $fatal(1,"simultaneous ACK erased event");
        $display("P2_18_PIR_ACK_EVENT_PRIORITY_PASS");

        // Clear counter, but not the independent pending bit.
        @(negedge clk);clear=1;
        @(negedge clk);clear=0;
        advance(2);
        if(count!=0 || !pending) $fatal(1,"counter clear affected pending");
        do_ack();
        if(pending || irq || count!=0)
            $fatal(1,"final ACK failed");
        $display("P2_18_PIR_COUNT_CLEAR_PASS");
        $display("P2_18_PIR_IRQ_CONTROLLER_RTL PASS");
        $finish;
    end
    initial begin #200000; $fatal(1,"TIMEOUT");end
endmodule

module tb_p2_18_axi_pir_irq;
    reg clk=0;always #5 clk=~clk;
    reg resetn=0,pir=0;
    reg [5:0] awaddr=0,araddr=0;
    reg awvalid=0,wvalid=0,arvalid=0;
    reg [31:0] wdata=0;
    reg [3:0] wstrb=4'hf;
    reg bready=0,rready=0;
    wire awready,wready,bvalid,arready,rvalid,irq;
    wire [31:0] rdata;
    wire [1:0] bresp,rresp;
    reg [31:0] result;
    p2_18_axi_pir_irq_v1_0 #(
        .WARMUP_CYCLES(64'd12),.FILTER_CYCLES(3)
    ) dut (
        .S_AXI_ACLK(clk),.S_AXI_ARESETN(resetn),
        .S_AXI_AWADDR(awaddr),.S_AXI_AWPROT(3'b0),
        .S_AXI_AWVALID(awvalid),.S_AXI_AWREADY(awready),
        .S_AXI_WDATA(wdata),.S_AXI_WSTRB(wstrb),
        .S_AXI_WVALID(wvalid),.S_AXI_WREADY(wready),
        .S_AXI_BRESP(bresp),.S_AXI_BVALID(bvalid),
        .S_AXI_BREADY(bready),.S_AXI_ARADDR(araddr),
        .S_AXI_ARPROT(3'b0),.S_AXI_ARVALID(arvalid),
        .S_AXI_ARREADY(arready),.S_AXI_RDATA(rdata),
        .S_AXI_RRESP(rresp),.S_AXI_RVALID(rvalid),
        .S_AXI_RREADY(rready),.pir_in_async(pir),.irq(irq)
    );
    task write_reg;
        input [5:0] address;
        input [31:0] data;
        input [3:0] strb;
        input [1:0] expect_resp;
        begin
            // AW then W deliberately arrive on separate cycles.
            @(negedge clk);awaddr=address;awvalid=1;
            @(posedge clk);if(!awready)$fatal(1,"AW not ready");
            @(negedge clk);awvalid=0;wdata=data;wstrb=strb;wvalid=1;
            @(posedge clk);if(!wready)$fatal(1,"W not ready");
            @(negedge clk);wvalid=0;
            wait(bvalid);#1;
            if(bresp!==expect_resp)
                $fatal(1,"BRESP address=%h expected=%h got=%h",
                    address,expect_resp,bresp);
            @(negedge clk);bready=1;
            @(negedge clk);bready=0;
        end
    endtask
    task read_reg;
        input [5:0] address;
        input [1:0] expect_resp;
        begin
            @(negedge clk);araddr=address;arvalid=1;
            @(posedge clk);if(!arready)$fatal(1,"AR not ready");
            @(negedge clk);arvalid=0;
            wait(rvalid);#1;
            result=rdata;
            if(rresp!==expect_resp)
                $fatal(1,"RRESP address=%h expected=%h got=%h",
                     address,expect_resp,rresp);
            @(negedge clk);rready=1;
            @(negedge clk);rready=0;
        end
    endtask
    task advance;
        input integer cycles;
        integer i;
        begin for(i=0;i<cycles;i=i+1)@(negedge clk);end
    endtask
    initial begin
        advance(4);resetn=1;advance(17);
        read_reg(6'h10,2'b00);
        if(result!==32'h00180001) $fatal(1,"IP ID");
        read_reg(6'h00,2'b00);
        if(result!==0 || irq) $fatal(1,"default CTRL not safe");
        $display("P2_18_AXI_ID_RESET_PASS");

        // A zero write-strobe must not change CTRL.
        write_reg(6'h00,32'h3,4'h0,2'b00);
        read_reg(6'h00,2'b00);
        if(result!==0) $fatal(1,"WSTRB ignored");
        write_reg(6'h00,32'h3,4'h1,2'b00);
        read_reg(6'h00,2'b00);
        if(result!==3) $fatal(1,"CTRL write failed");
        $display("P2_18_AXI_WSTRB_CTRL_PASS");

        @(negedge clk);pir=1;advance(12);
        if(!irq)$fatal(1,"IRQ not asserted");
        read_reg(6'h08,2'b00);
        if(result[3:0]!==4'b1111)$fatal(1,"STATUS incorrect %h",result);
        read_reg(6'h0c,2'b00);
        if(result!==1) $fatal(1,"count missing");
        $display("P2_18_AXI_IRQ_STATUS_COUNT_PASS");

        write_reg(6'h00,32'h1,4'h1,2'b00);
        advance(3);if(irq!==0) $fatal(1,"mask ineffective");
        read_reg(6'h08,2'b00);
        if(result[2]!==1) $fatal(1,"mask erased pending");
        write_reg(6'h00,32'h3,4'h1,2'b00);
        advance(2);if(!irq) $fatal(1,"unmask lost pending");
        write_reg(6'h04,32'h1,4'h1,2'b00);advance(3);
        if(irq!==0) $fatal(1,"ACK failed");
        read_reg(6'h0c,2'b00);
        if(result!==1) $fatal(1,"ACK cleared counter");
        $display("P2_18_AXI_IRQ_MASK_ACK_PASS");

        write_reg(6'h04,32'h2,4'h1,2'b00);advance(3);
        read_reg(6'h0c,2'b00);
        if(result!==0) $fatal(1,"COUNT_CLEAR failed");
        write_reg(6'h0c,32'hffff,4'hf,2'b10);
        read_reg(6'h04,2'b10);
        read_reg(6'h15,2'b10);
        $display("P2_18_AXI_ERROR_RESP_PASS");
        $display("P2_18_AXI_PIR_IRQ_PASS");
        $finish;
    end
    initial begin #250000;$fatal(1,"TIMEOUT");end
endmodule

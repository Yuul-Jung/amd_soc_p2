01과제 개요

제2부 12장의 stepper_core.v와 AXI v1.0을 변경하지 않고, 가감속 기능을 가진 신규 stepper_ramp_core.v 및 p2_12_axi_stepper_v1_1.v를 추가한다. 이로써 고정 주기 스텝 구동에서 저속 시작 → 가속 → 정속 → 감속 → 정지로 확장한다.

Vivado 실제 합성/타이밍, Vitis 타깃 C, ULN2003+모터 실물 가감속, 안전한 최대속도.

02가감속 원리와 타이밍

모터 코일 여자 상태 자체는 이전과 동일하다. 다음 스텝이 발생할 때까지 기다리는 active_period만 구간별로 변화한다.

위의 8→6→4→2→2→2→2→4→6→8은 축척된 RTL 테스트 시계 수이다. 실제 100MHz 클록에서는 설정값을 밀리초에서 tick으로 변환한다. 주기가 짧아질수록 스텝 주파수는 증가하므로 가속 상태에 해당한다.

파라미터역할실제 100MHz 예제
START_PERIOD초기 느린 스텝 주기10ms → 1,000,000 tick
CRUISE_PERIOD정속 구간의 짧은 주기4ms → 400,000 tick
RAMP_STEPS가속·감속 각 구간의 스텝 수12
DELTA_PERIOD한 스텝마다 주기를 줄이거나 늘리는 크기50,000 tick
STEP_COUNT목표 스텝128

조건: 시작 주기와 정속 주기의 차이는 DELTA_PERIOD × RAMP_STEPS로 정확하게 나누어져야 한다. 새로운 stepper_ramp_make_config() 함수가 이를 검사한다. 가감속 구간은 겹치지 않도록 2 × RAMP_STEPS < STEP_COUNT로 제한한다.

명령·출력
`timescale 1ns/1ps
// P2-12.4: 기존 stepper_core.v를 보존하는 가감속 확장 코어.
// (추가) 시작 저속 -> 가속 -> 정속 -> 감속 -> 정지.
// (주의) 가감속은 스텝 간 간격을 계단식으로 조절하는 선형 주기 프로파일.
// (주의) 속도/가속도가 실제 기계의 연속 선형 가속도가 되지는 않는다.
// Clock Domain: clk 단일 도메인, AXI와 같은 클록 사용을 전제로 함.
module stepper_ramp_core #(
    parameter [31:0] MIN_STEP_CYCLES = 32'd200000
)(
    input wire clk, input wire resetn,
    input wire start_pulse, input wire stop_pulse,
    input wire [31:0] cfg_steps,
    input wire [31:0] cfg_start_period,
    input wire [31:0] cfg_cruise_period,
    input wire [31:0] cfg_delta_period,
    input wire [31:0] cfg_ramp_steps,
    input wire cfg_reverse, input wire cfg_half_step,
    output wire [3:0] coil_o,
    output reg busy, output reg done_pulse, output reg error_pulse,
    output reg [31:0] steps_done,
    output reg [31:0] active_period
);
    reg [31:0] target_steps;
    reg [31:0] cruise_period;
    reg [31:0] start_period;
    reg [31:0] delta_period;
    reg [31:0] ramp_steps;
    reg [31:0] ticks_left;
    reg [2:0] phase;
    reg reverse_q, half_q;
    reg [3:0] coils_q;

    function [3:0] pattern;
        input [2:0] idx;
        input       half;
        begin
            if(half) begin
                case(idx)
                3'd0:pattern=4'b0001;
                3'd1:pattern=4'b0011;
                3'd2:pattern=4'b0010;
                3'd3:pattern=4'b0110;
                3'd4:pattern=4'b0100;
                3'd5:pattern=4'b1100;
                3'd6:pattern=4'b1000;
                default:pattern=4'b1001;
                endcase
            end else begin
                case(idx[1:0])
                2'd0:pattern=4'b0011;
                2'd1:pattern=4'b0110;
                2'd2:pattern=4'b1100;
                default:pattern=4'b1001;
                endcase
            end
        end
    endfunction

    // 실제 코일 출력은 STOP/RESET에서 즉시 차단.
    assign coil_o = (resetn && !stop_pulse) ? coils_q : 4'b0000;

    // cfg_start_period = cfg_cruise_period + cfg_delta_period * cfg_ramp_steps
    // 일치하도록 C Host 단계에서 사전 검증한다. RTL은 감산/가산 포화 적용.
    always @(posedge clk) begin
        if(!resetn) begin
            busy <= 1'b0;
            done_pulse <= 1'b0;
            error_pulse <= 1'b0;
            steps_done <= 32'd0;
            active_period <= MIN_STEP_CYCLES;
            target_steps <= 32'd0;
            cruise_period <= MIN_STEP_CYCLES;
            start_period <= MIN_STEP_CYCLES;
            delta_period <= 32'd0;
            ramp_steps <= 32'd0;
            ticks_left <= 32'd0;
            phase <= 3'd0;
            reverse_q <= 1'b0;
            half_q <= 1'b1;
            coils_q <= 4'b0000;
        end else begin
            done_pulse <= 1'b0;
            error_pulse <= 1'b0;
            if(stop_pulse) begin
                busy <= 1'b0;
                coils_q <= 4'b0000;
                ticks_left <= 32'd0;
            end else if(start_pulse) begin
                // 2*R < N 필수: 가속/감속 구간이 겹치지 않도록 제한.
                if(busy || cfg_steps < 3 ||
                   cfg_ramp_steps == 0 ||
                   cfg_ramp_steps > ((cfg_steps - 32'd1) >> 1) ||
                   cfg_start_period <= cfg_cruise_period ||
                   cfg_cruise_period < MIN_STEP_CYCLES ||
                   cfg_delta_period == 0 ||
                   cfg_start_period < cfg_delta_period) begin
                    error_pulse <= 1'b1;
                end else begin
                    busy <= 1'b1;
                    steps_done <= 32'd0;
                    target_steps <= cfg_steps;
                    start_period <= cfg_start_period;
                    cruise_period <= cfg_cruise_period;
                    delta_period <= cfg_delta_period;
                    ramp_steps <= cfg_ramp_steps;
                    active_period <= cfg_start_period;
                    ticks_left <= cfg_start_period - 32'd1;
                    reverse_q <= cfg_reverse;
                    half_q <= cfg_half_step;
                    phase <= 3'd0;
                    coils_q <= 4'b0000;
                end
            end else if(busy) begin
                if(ticks_left != 0) begin
                    ticks_left <= ticks_left - 32'd1;
                end else begin
                    coils_q <= pattern(phase,half_q);
                    steps_done <= steps_done + 32'd1;
                    // 다음 step 주기값이 갱신되어도 지금 수행한 step은 유지.
                    if(steps_done + 32'd1 == target_steps) begin
                        busy <= 1'b0;
                        done_pulse <= 1'b1;
                    end else begin
                        if(reverse_q)
                            phase <= half_q ? (phase - 3'd1)
                                            : {1'b0,(phase[1:0] - 2'd1)};
                        else
                            phase <= half_q ? (phase + 3'd1)
                                            : {1'b0,(phase[1:0] + 2'd1)};
                        // next_index = steps_done+1, 0부터 시작.
                        if(steps_done + 32'd1 < ramp_steps) begin
                            // 가속 구간: period를 작은 값으로.
                            if(active_period > cruise_period &&
                               active_period - cruise_period > delta_period)
                                active_period <= active_period - delta_period;
                            else
                                active_period <= cruise_period;
                        end else if(steps_done + 32'd1 >= target_steps-ramp_steps) begin
                            // 감속 구간: period를 큰 값으로.
                            if(active_period < start_period &&
                               start_period - active_period > delta_period)
                                active_period <= active_period + delta_period;
                            else
                                active_period <= start_period;
                        end else begin
                            active_period <= cruise_period;
                        end
                        // 아래 구간에서 update된 active_period는 NBA이므로
                        // 타이머에는 다음 주기 길이가 필요하다.
                        // 다음 주기 tick 로드는 분기별로 직접 계산한다.
                        if(steps_done + 32'd1 < ramp_steps) begin
                            ticks_left <= ((active_period > cruise_period &&
                               active_period - cruise_period > delta_period)
                                          ? active_period-delta_period
                                          : cruise_period) - 32'd1;
                        end else if(steps_done + 32'd1 >= target_steps-ramp_steps) begin
                            ticks_left <= ((active_period < start_period-delta_period)
                                          ? active_period+delta_period
                                          : start_period) - 32'd1;
                        end else begin
                            ticks_left <= cruise_period - 32'd1;
                        end
                    end
                end
            end else begin
                coils_q <= 4'b0000;
            end
        end
    end
endmodule

`timescale 1ns/1ps
// (추가) P2-12.4 전용 가감속 AXI4-Lite v1_1: 기존 v1_0 변경 없음.
// 확장 레지스터 0x20 CRUISE, 0x24 DELTA, 0x28 RAMP_STEPS, 0x2C ACTIVE_PERIOD.
// Original P2-01~10 AXI IPs are not modified.
// Registers: 0x00 STEPS, 0x04 PERIOD_TICKS, 0x08 CONTROL,
//            0x0C CMD, 0x10 STATUS, 0x14 DONE_STEPS,
//            0x18 ID, 0x1C MIN_STEP_TICKS.
// CONTROL bit0=reverse, bit1=half_step. CMD bit0=START, bit1=STOP.
// STATUS bit0=BUSY, bit1=DONE_STICKY, bit2=ERROR_STICKY.
module p2_12_axi_stepper_v1_1 #(
    parameter integer C_S_AXI_DATA_WIDTH = 32,
    parameter integer C_S_AXI_ADDR_WIDTH = 6,
    parameter [31:0] MIN_STEP_CYCLES = 32'd200000,
    parameter [31:0] RESET_STEP_PERIOD = 32'd500000
)(
    input wire S_AXI_ACLK, input wire S_AXI_ARESETN,
    input wire [C_S_AXI_ADDR_WIDTH-1:0] S_AXI_AWADDR,
    input wire [2:0] S_AXI_AWPROT,
    input wire S_AXI_AWVALID, output wire S_AXI_AWREADY,
    input wire [C_S_AXI_DATA_WIDTH-1:0] S_AXI_WDATA,
    input wire [(C_S_AXI_DATA_WIDTH/8)-1:0] S_AXI_WSTRB,
    input wire S_AXI_WVALID, output wire S_AXI_WREADY,
    output reg [1:0] S_AXI_BRESP, output reg S_AXI_BVALID,
    input wire S_AXI_BREADY,
    input wire [C_S_AXI_ADDR_WIDTH-1:0] S_AXI_ARADDR,
    input wire [2:0] S_AXI_ARPROT,
    input wire S_AXI_ARVALID, output wire S_AXI_ARREADY,
    output reg [C_S_AXI_DATA_WIDTH-1:0] S_AXI_RDATA,
    output reg [1:0] S_AXI_RRESP, output reg S_AXI_RVALID,
    input wire S_AXI_RREADY,
    output wire [3:0] stepper_coil_o
);
    localparam [1:0] OKAY = 2'b00, SLVERR = 2'b10;
    localparam [31:0] ID = 32'h000C0002;
    reg [31:0] cfg_steps;
    reg [31:0] cfg_period;
    // (추가) 가감속을 위한 확장 설정값.
    reg [31:0] cfg_cruise_period, cfg_delta_period, cfg_ramp_steps;
    reg [31:0] cfg_control;
    reg start_strobe, stop_strobe;
    wire busy, done_pulse, error_pulse;
    wire [31:0] steps_done;
    wire [31:0] active_period;
    reg done_sticky, error_sticky;
    reg aw_hold, w_hold;
    reg [C_S_AXI_ADDR_WIDTH-1:0] saved_awaddr;
    reg [31:0] saved_wdata;
    reg [3:0] saved_wstrb;
    reg [31:0] merged;
    integer b;

    stepper_ramp_core #(.MIN_STEP_CYCLES(MIN_STEP_CYCLES)) u_core (
        .clk(S_AXI_ACLK), .resetn(S_AXI_ARESETN),
        .start_pulse(start_strobe), .stop_pulse(stop_strobe),
        .cfg_steps(cfg_steps), .cfg_start_period(cfg_period),
        .cfg_cruise_period(cfg_cruise_period),
        .cfg_delta_period(cfg_delta_period),
        .cfg_ramp_steps(cfg_ramp_steps),
        .cfg_reverse(cfg_control[0]),.cfg_half_step(cfg_control[1]),
        .coil_o(stepper_coil_o), .busy(busy),
        .done_pulse(done_pulse),.error_pulse(error_pulse),
        .steps_done(steps_done),.active_period(active_period)
    );
    // 각 AW/W 채널 버퍼 한 건 + 응답 한 건만 보관.
    assign S_AXI_AWREADY = S_AXI_ARESETN && !aw_hold && !S_AXI_BVALID;
    assign S_AXI_WREADY  = S_AXI_ARESETN && !w_hold  && !S_AXI_BVALID;
    assign S_AXI_ARREADY = S_AXI_ARESETN && !S_AXI_RVALID;

    always @(posedge S_AXI_ACLK) begin
        if (!S_AXI_ARESETN) begin
            cfg_steps     <= 32'd128;
            cfg_period    <= RESET_STEP_PERIOD;
            cfg_cruise_period <= 32'd200000;
            cfg_delta_period <= 32'd100000;
            cfg_ramp_steps <= 32'd3;
            cfg_control   <= 32'd2; // default HALF, forward
            aw_hold       <= 1'b0;
            w_hold        <= 1'b0;
            saved_awaddr  <= {C_S_AXI_ADDR_WIDTH{1'b0}};
            saved_wdata   <= 32'd0;
            saved_wstrb   <= 4'd0;
            S_AXI_BVALID  <= 1'b0;
            S_AXI_BRESP   <= OKAY;
            start_strobe  <= 1'b0;
            stop_strobe   <= 1'b0;
            done_sticky   <= 1'b0;
            error_sticky  <= 1'b0;
        end else begin
            start_strobe <= 1'b0;
            stop_strobe  <= 1'b0;
            if (done_pulse) done_sticky <= 1'b1;
            if (error_pulse) error_sticky <= 1'b1;
            if (S_AXI_AWVALID && S_AXI_AWREADY) begin
                saved_awaddr <= S_AXI_AWADDR;
                aw_hold <= 1'b1;
            end
            if (S_AXI_WVALID && S_AXI_WREADY) begin
                saved_wdata <= S_AXI_WDATA;
                saved_wstrb <= S_AXI_WSTRB;
                w_hold <= 1'b1;
            end
            if (S_AXI_BVALID && S_AXI_BREADY)
                S_AXI_BVALID <= 1'b0;
            if (aw_hold && w_hold && !S_AXI_BVALID) begin
                aw_hold <= 1'b0;
                w_hold <= 1'b0;
                S_AXI_BVALID <= 1'b1;
                S_AXI_BRESP <= OKAY;
                case (saved_awaddr)
                6'h00: begin
                    merged=cfg_steps;
                    for (b=0;b<4;b=b+1)
                        if(saved_wstrb[b])
                            merged[8*b +:8]=saved_wdata[8*b +:8];
                    cfg_steps <= merged;
                end
                6'h04: begin
                    merged=cfg_period;
                    for (b=0;b<4;b=b+1)
                        if(saved_wstrb[b])
                            merged[8*b +:8]=saved_wdata[8*b +:8];
                    cfg_period <= merged;
                end
                6'h08: begin
                    merged=cfg_control;
                    for (b=0;b<4;b=b+1)
                        if(saved_wstrb[b])
                            merged[8*b +:8]=saved_wdata[8*b +:8];
                    cfg_control <= (merged & 32'h3);
                end
                6'h0C: begin
                    // 반드시 낮은 바이트 스트로브와 명령 비트가 있어야 함.
                    if (!saved_wstrb[0]) begin
                        S_AXI_BRESP <= SLVERR;
                    end else if (saved_wdata[1] && !saved_wdata[0]) begin
                        stop_strobe <= 1'b1;
                    end else if (saved_wdata[0] && !saved_wdata[1]) begin
                        if (busy || start_strobe || cfg_steps < 3 ||
                            cfg_ramp_steps == 0 ||
                            cfg_ramp_steps > ((cfg_steps - 32'd1) >> 1) ||
                            cfg_period <= cfg_cruise_period ||
                            cfg_cruise_period < MIN_STEP_CYCLES ||
                            cfg_delta_period == 0) begin
                            S_AXI_BRESP <= SLVERR;
                            error_sticky <= 1'b1;
                        end else begin
                            start_strobe <= 1'b1;
                            done_sticky <= 1'b0;
                            error_sticky <= 1'b0;
                        end
                    end else S_AXI_BRESP <= SLVERR;
                end
                // (추가) 확장 Configuration: 저장용 Shadow, START 때 래치.
                6'h20: begin
                    merged=cfg_cruise_period;
                    for(b=0;b<4;b=b+1)
                        if(saved_wstrb[b])
                            merged[8*b +:8]=saved_wdata[8*b +:8];
                    cfg_cruise_period<=merged;
                end
                6'h24: begin
                    merged=cfg_delta_period;
                    for(b=0;b<4;b=b+1)
                        if(saved_wstrb[b])
                            merged[8*b +:8]=saved_wdata[8*b +:8];
                    cfg_delta_period<=merged;
                end
                6'h28: begin
                    merged=cfg_ramp_steps;
                    for(b=0;b<4;b=b+1)
                        if(saved_wstrb[b])
                            merged[8*b +:8]=saved_wdata[8*b +:8];
                    cfg_ramp_steps<=merged;
                end
                default: S_AXI_BRESP <= SLVERR;
                endcase
            end
        end
    end

    always @(posedge S_AXI_ACLK) begin
        if (!S_AXI_ARESETN) begin
            S_AXI_RVALID <= 1'b0;
            S_AXI_RDATA  <= 32'd0;
            S_AXI_RRESP  <= OKAY;
        end else begin
            if (S_AXI_RVALID && S_AXI_RREADY)
                S_AXI_RVALID <= 1'b0;
            if (S_AXI_ARVALID && S_AXI_ARREADY) begin
                S_AXI_RVALID <= 1'b1;
                S_AXI_RRESP <= OKAY;
                case (S_AXI_ARADDR)
                6'h00: S_AXI_RDATA <= cfg_steps;
                6'h04: S_AXI_RDATA <= cfg_period;
                6'h08: S_AXI_RDATA <= cfg_control;
                6'h0C: S_AXI_RDATA <= 32'd0;
                6'h10: S_AXI_RDATA <= {29'd0,error_sticky,done_sticky,busy};
                6'h14: S_AXI_RDATA <= steps_done;
                6'h18: S_AXI_RDATA <= ID;
                6'h1C: S_AXI_RDATA <= MIN_STEP_CYCLES;
                6'h20: S_AXI_RDATA <= cfg_cruise_period;
                6'h24: S_AXI_RDATA <= cfg_delta_period;
                6'h28: S_AXI_RDATA <= cfg_ramp_steps;
                6'h2C: S_AXI_RDATA <= active_period;
                default: begin
                    S_AXI_RDATA <= 32'd0;
                    S_AXI_RRESP <= SLVERR;
                end
                endcase
            end
        end
    end
endmodule

#include "stepper_ramp_logic.h"
#include <stdint.h>
#include <limits.h>
int stepper_ramp_make_config(uint32_t clock_hz, uint32_t steps,
    uint32_t start_ms, uint32_t cruise_ms, uint32_t ramp_steps,
    int reverse, int half_step, stepper_ramp_cfg_t *out)
{
    uint64_t slow,fast,diff;
    if(!out || clock_hz < 1000000u || clock_hz > 200000000u ||
       steps < 3u || steps > 100000u ||
       ramp_steps == 0u || ramp_steps > (steps-1u)/2u ||
       start_ms < 2u || start_ms > 100u ||
       cruise_ms < 2u || cruise_ms >= start_ms ||
       (reverse != 0 && reverse != 1) ||
       (half_step != 0 && half_step != 1)) return -1;
    slow=((uint64_t)clock_hz * start_ms) / 1000u;
    fast=((uint64_t)clock_hz * cruise_ms) / 1000u;
    if(slow > UINT32_MAX || fast > UINT32_MAX || fast < 2u)
        return -1;
    diff=slow-fast;
    // 매 스텝 간격 차이가 정수 tick으로 정확하게 일치해야 한다.
    if(diff < ramp_steps || diff % ramp_steps != 0u)
        return -1;
    if(diff/ramp_steps > UINT32_MAX)
        return -1;
    out->steps=steps;
    out->start_ticks=(uint32_t)slow;
    out->cruise_ticks=(uint32_t)fast;
    out->delta_ticks=(uint32_t)(diff/ramp_steps);
    out->ramp_steps=ramp_steps;
    out->control=(uint32_t)reverse | ((uint32_t)half_step << 1);
    return 0;
}

// P2-12.4 Vitis Ramp AXI v1.1 사용 예시.
// 원본 P2-12 V1.0 AXI IP 주소를 사용하지 않는다.
// 실제 하드웨어에서 별도 P2-12.4 IP v1.1을 패키징한 경우에만 사용한다.
#include "xparameters.h"
#include "xil_io.h"
#include "xil_printf.h"
#include "../sw/stepper_ramp_logic.h"

#ifndef P2_12_RAMP_BASEADDR
#error "Set P2_12_RAMP_BASEADDR using actual XSA Address Editor"
#endif
#ifndef P2_12_AXI_CLOCK_HZ
#error "Set P2_12_AXI_CLOCK_HZ using actual Block Design clock"
#endif

#define R_STEPS      0x00u
#define R_START      0x04u
#define R_CONTROL    0x08u
#define R_COMMAND    0x0Cu
#define R_STATUS     0x10u
#define R_DONE       0x14u
#define R_ID         0x18u
#define R_MIN_PERIOD 0x1Cu
#define R_CRUISE     0x20u
#define R_DELTA      0x24u
#define R_RAMP_STEPS 0x28u
#define R_ACTIVE     0x2Cu

int p2_12_4_start_ramp_example(void)
{
    const UINTPTR base = (UINTPTR)P2_12_RAMP_BASEADDR;
    stepper_ramp_cfg_t c;
    uint32_t status;
    // 예제: 128 step, 10 ms → 4 ms, 가속 12 step, 감속 12 step.
    if (stepper_ramp_make_config(
            P2_12_AXI_CLOCK_HZ, 128u, 10u, 4u, 12u, 0, 1, &c) != 0) {
        xil_printf("ERR ramp config\r\n");
        return -1;
    }
    if (Xil_In32(base + R_ID) != 0x000C0002u ||
        c.cruise_ticks < Xil_In32(base + R_MIN_PERIOD)) {
        xil_printf("ERR wrong IP or too-fast ramp\r\n");
        return -1;
    }
    status = Xil_In32(base + R_STATUS);
    if (status & 1u) {
        xil_printf("ERR busy\r\n");
        return -1;
    }
    Xil_Out32(base + R_STEPS, c.steps);
    Xil_Out32(base + R_START, c.start_ticks);
    Xil_Out32(base + R_CRUISE, c.cruise_ticks);
    Xil_Out32(base + R_DELTA, c.delta_ticks);
    Xil_Out32(base + R_RAMP_STEPS, c.ramp_steps);
    Xil_Out32(base + R_CONTROL, c.control);
    Xil_Out32(base + R_COMMAND, 1u); // START: Vitis register write, not a completion
    xil_printf("RAMP issued. Poll STATUS, ACTIVE_PERIOD, DONE.\r\n");
    return 0;
}

void p2_12_4_stop_ramp_example(void)
{
    const UINTPTR base = (UINTPTR)P2_12_RAMP_BASEADDR;
    Xil_Out32(base + R_COMMAND, 2u); // STOP
}

06Vivado/Vitis 실제 확장 절차

  1. 12.1에서 만든 기존 프로젝트를 다시 별도로 Save As 하여 soc_ip_design_12_4로 만든다.
  2. IP Packager에서 stepper_ramp_core.v와 p2_12_axi_stepper_v1_1.v를 패키징하고 신규 v1.1 IP와 기존 v1.0이 구분되게 한다.
  3. 기존 Stepper AXI v1.0을 직접 삭제하거나 덮어쓰지 않고 복제 프로젝트에서 신버전 인스턴스를 배치한다. 주소가 겹치지 않는지 확인한다.
  4. IP의 stepper_coil_o[3:0]는 ULN2003의 기존 실물 신호 배치를 따르되 출력 소유권을 이중으로 연결하지 않는다.
  5. 새 XSA와 BSP로 Vitis 재설정 후 12_4_ramp_vitis_example.c에서 요구하는 BaseAddr/Clock을 실제 환경 값으로 제공한다.
  6. 실물은 128스텝, START 10ms, CRUISE 4ms, RAMP_STEPS 12부터 무부하로 시작한다. 가속 중 탈조·전원 강하·모터 발열을 실제 점검한다.

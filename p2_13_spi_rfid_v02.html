01개요

SPI(Serial Peripheral Interface, 에스피아이, 동기식 직렬 주변장치 인터페이스)는 SCK(클록), MOSI(마스터 송신), MISO(마스터 수신), NSS(슬레이브 선택) 4개 신호로 통신한다. RC522 보드의 MFRC522는 SPI 슬레이브로 동작하며 FPGA가 클록과 명령을 공급한다.

SPI Mode 0(클록 유휴 LOW, 상승 에지 수신), 16비트 레지스터 프레임, MFRC522 VersionReg(0x37) 읽기, 동작 자체를 확인하는 RTL 자가확인 테스트벤치.

미완료: 실제 RC522의 안테나 설정, REQA/WUPA, 안티콜리전·UID 수집, IRQ 처리, 실제 AXI 인스턴스, Vitis 타깃 구동, Vivado 합성·타이밍, 카드 실물 감지. VersionReg 값을 읽을 수 있다고 카드 인식까지 성공한 것은 아니다.

02전체 기능 구조

mfrc522_version_system.v은 두 독립 모듈을 연결하는 Wrapper이며, 사용자 Top이나 Basys3 핀 이름에 종속되지 않는다.

05RTL 동작 구조 및 테스트벤치

TX 알고리즘: ① 사용자 명령 수신 → ② 주소/데이터 16비트 레지스터 로드 → ③ NSS LOW → ④ SCK 상승에서 MISO 수신 → ⑤ SCK 하강에서 MOSI 변경 → ⑥ 총 16비트 전송 → ⑦ NSS HIGH 및 1클록 rsp_valid.

SPI SCK는 출력 신호로 생성하되 내부 상태 머신은 하나의 FPGA 시스템 클록만 사용한다. 자체 별도 클록 도메인 생성 없이 클록 디바이더 카운터로 시점을 선택한다.

rtl/mfrc522_spi_master.v · VERILOG · 소스
명령·출력
`timescale 1ns/1ps
// P2-13 · MFRC522 16-bit SPI register transaction.
// CPOL=0, CPHA=0, MSB first; CS asserted throughout 2 bytes.
// CMD handshakes on valid && ready; done pulses after 16 rising edges.
// Input clk only, no generated FPGA clock domain. SCK is output GPIO.
// SPI bit rate = clk / (2 * CLK_DIV). e.g. 100MHz/20 = 5MHz.
// No RST output, antenna setup or RFID card/UID procedure here.
module mfrc522_spi_master #(
    parameter integer CLK_DIV = 10
)(
    input wire clk, input wire resetn,
    input wire cmd_valid, output wire cmd_ready,
    input wire cmd_read,
    input wire [5:0] cmd_addr,
    input wire [7:0] cmd_wdata,
    output reg [7:0] rsp_rdata,
    output reg rsp_valid,
    output reg spi_sck,
    output reg spi_mosi,
    input wire spi_miso,
    output reg spi_nss
);
    localparam [1:0] IDLE=0, LOW=1, HIGH=2, FINISH=3;
    reg [1:0] state;
    reg [15:0] tx_word;
    reg [7:0] rx_shift;
    reg [4:0] bit_pos;
    reg [31:0] div_count;
    assign cmd_ready = (state == IDLE);

    always @(posedge clk) begin
        if (!resetn) begin
            state <= IDLE; tx_word <= 16'd0; rx_shift <= 8'd0;
            bit_pos <= 5'd0; div_count <= 32'd0;
            spi_sck <= 1'b0; spi_mosi <= 1'b0; spi_nss <= 1'b1;
            rsp_rdata <= 8'd0; rsp_valid <= 1'b0;
        end else begin
            rsp_valid <= 1'b0;
            case (state)
            IDLE: begin
                spi_sck <= 1'b0; spi_nss <= 1'b1;
                if (cmd_valid) begin
                    // First byte [R/W, six-bit register address, 0].
                    // For reads: MOSI second byte is dummy 8'h00.
                    tx_word <= {{cmd_read,cmd_addr,1'b0},
                                (cmd_read ? 8'h00 : cmd_wdata)};
                    spi_mosi <= cmd_read; // first outgoing bit
                    spi_nss <= 1'b0;
                    bit_pos <= 5'd15; div_count <= 32'd0;
                    rx_shift <= 8'd0;
                    state <= LOW;
                end
            end
            LOW: begin
                // SCK low for CLK_DIV FPGA cycles; MOSI stable before rising edge.
                if(div_count == CLK_DIV-1) begin
                    div_count <= 32'd0;
                    spi_sck <= 1'b1;
                    if(bit_pos < 5'd8)
                        rx_shift <= {rx_shift[6:0], spi_miso};
                    state <= HIGH;
                end else div_count <= div_count + 32'd1;
            end
            HIGH: begin
                if(div_count == CLK_DIV-1) begin
                    div_count <= 32'd0;
                    spi_sck <= 1'b0;
                    if(bit_pos == 5'd0) begin
                        state <= FINISH;
                    end else begin
                        bit_pos <= bit_pos - 5'd1;
                        // Advance output bit at falling edge, sample next rising edge.
                        spi_mosi <= tx_word[bit_pos-5'd1];
                        state <= LOW;
                    end
                end else div_count <= div_count + 32'd1;
            end
            FINISH: begin
                spi_nss <= 1'b1; spi_sck <= 1'b0;
                rsp_rdata <= rx_shift;
                rsp_valid <= 1'b1;
                state <= IDLE;
            end
            default: state <= IDLE;
            endcase
        end
    end
endmodule

// P2-13 · VersionReg(0x37) read once per request, independent controller.
// No card detection or UID authentication performed by this module.
module mfrc522_version_probe (
    input wire clk, input wire resetn,
    input wire start_pulse,
    output reg [7:0] version,
    output reg done_pulse, output reg busy,
    output wire cmd_valid, input wire cmd_ready,
    output wire cmd_read, output wire [5:0] cmd_addr,
    output wire [7:0] cmd_wdata,
    input wire [7:0] rsp_rdata, input wire rsp_valid
);
    localparam [1:0] WAIT=0, SEND=1, RESPONSE=2;
    reg [1:0] state;
    assign cmd_valid=(state == SEND);
    assign cmd_read=1'b1;
    assign cmd_addr=6'h37;
    assign cmd_wdata=8'h00;
    always @(posedge clk) begin
        if(!resetn) begin
            state<=WAIT; version<=8'd0; done_pulse<=0; busy<=0;
        end else begin
            done_pulse<=0;
            case(state)
            WAIT: if(start_pulse) begin busy<=1; state<=SEND; end
            SEND: if(cmd_ready) state<=RESPONSE;
            RESPONSE: if(rsp_valid) begin
                version<=rsp_rdata; done_pulse<=1; busy<=0; state<=WAIT;
            end
            default: begin state<=WAIT;busy<=0;end
            endcase
        end
    end
endmodule

`timescale 1ns/1ps
module tb_mfrc522_spi_master;
    reg clk=0; always #5 clk=~clk;
    reg resetn=0,cmd_valid=0,cmd_read=0;
    reg [5:0] addr=0;
    reg [7:0] wdata=0;
    wire cmd_ready,rsp_valid,sck,mosi,nss;
    wire [7:0] rdata;
    reg miso=0;
    reg [15:0] actual_tx;
    reg [15:0] response_pattern=16'h00A5;
    integer rise_count;
    integer done_count=0;
    mfrc522_spi_master #(.CLK_DIV(3)) dut(
        .clk(clk),.resetn(resetn),.cmd_valid(cmd_valid),
        .cmd_ready(cmd_ready),.cmd_read(cmd_read),.cmd_addr(addr),
        .cmd_wdata(wdata),.rsp_rdata(rdata),.rsp_valid(rsp_valid),
        .spi_sck(sck),.spi_mosi(mosi),.spi_miso(miso),.spi_nss(nss)
    );
    // Slave prepares MISO on falling edge; first response bit already
    // present before first rising edge (first byte discarded by master).
    always @(negedge sck) begin
        if (!nss && rise_count < 16)
            miso = response_pattern[15-rise_count];
    end
    always @(posedge sck) begin
        if(nss) $fatal(1,"NSS not low during SCK");
        actual_tx[15-rise_count]=mosi;
        rise_count=rise_count+1;
    end
    always @(posedge rsp_valid) begin
        done_count=done_count+1;
        if(!nss || sck) $fatal(1,"response while bus active");
    end
    task transmit;
        input rd;
        input [5:0] address;
        input [7:0] data;
        input [15:0] expect_tx;
        input [7:0] expect_rx;
        begin
            @(negedge clk);
            cmd_read=rd;addr=address;wdata=data;cmd_valid=1;
            rise_count=0;actual_tx=0;
            response_pattern={8'h00,expect_rx};
            miso=0;
            @(negedge clk);cmd_valid=0;
            wait(rsp_valid);
            #1;
            if(rise_count!=16) $fatal(1,"expected 16 SCK rising edges, got %0d",rise_count);
            if(actual_tx !== expect_tx)
                $fatal(1,"MOSI expected=%04h got=%04h",expect_tx,actual_tx);
            if(rdata !== expect_rx)
                $fatal(1,"MISO expected=%02h got=%02h",expect_rx,rdata);
            @(negedge clk);
            if(!cmd_ready || !nss) $fatal(1,"not idle after response");
        end
    endtask
    initial begin
        repeat(3) @(negedge clk);resetn=1;
        if(nss!==1 || sck!==0) $fatal(1,"idle polarity");
        // Read VersionReg 0x37 → address byte EE, dummy 00.
        transmit(1,6'h37,8'hcc,16'hEE00,8'h92);
        $display("P2_13_SPI_VERSION_REGISTER_READ PASS");
        // Write arbitrary test register 0x14: addr=0x28, value=0x5A.
        transmit(0,6'h14,8'h5A,16'h285A,8'hA5);
        $display("P2_13_SPI_REGISTER_WRITE PASS");
        // Busy command request must be ignored.
        if(done_count!=2) $fatal(1,"unexpected extra response");
        $display("P2_13_SPI_MASTER_RTL PASS");
        $finish;
    end
    initial begin #200000; $fatal(1,"TIMEOUT");end
endmodule

13.3 제한형 4바이트 UID 프로토콜 C 실습

MFRC522 추상 Register I/O를 통해 REQA → ATQA → 0x93 0x20 Anticollision → UID[4] + BCC를 구현한다. C 호스트 테스트는 가짜 칩 모델에서 UID, BCC 오류, 충돌 확장 바이트(0x88), 응답 타임아웃을 확인한다. 이 코드는 아직 실제 FPGA·카드와 연결되어 있지 않으며, 카드 선택(SELECT)·7/10바이트 UID·다중 카드 충돌 복구는 구현하지 않았다.

sw/mfrc522_picc.h · C · 소스
C
#ifndef P2_13_MFRC522_PICC_H
#define P2_13_MFRC522_PICC_H
#include <stdint.h>
#include <stddef.h>

// 보드에 종속되지 않는 Register I/O 추상 인터페이스.
// 반환: 0 = 성공, 0이 아니면 SPI/버스 접근 오류.
typedef int (*rc522_read_fn)(void *ctx, uint8_t reg, uint8_t *value);
typedef int (*rc522_write_fn)(void *ctx, uint8_t reg, uint8_t value);
typedef struct {
    void *ctx;
    rc522_read_fn read;
    rc522_write_fn write;
} rc522_bus_t;

typedef enum {
    RC522_OK = 0,
    RC522_IO_ERROR = -1,
    RC522_TIMEOUT = -2,
    RC522_PROTOCOL_ERROR = -3,
    RC522_COLLISION = -4,
    RC522_UNSUPPORTED_UID = -5
} rc522_status_t;

// MFRC522 전원이 켜져 있고 Mode/Timer/TxControl 초기화가 끝난 상태에서만.
// REQA(7-bit) → ATQA(2B) → CL1 Anticollision(5B UID/BCC).
// 오직 충돌 없는 단일 4바이트 UID 카드 대상으로 실습.
rc522_status_t rc522_read_uid4(const rc522_bus_t *bus, uint8_t uid[4],
                                uint8_t atqa[2], uint32_t max_polls);
#endif

sw/mfrc522_picc.c · C · 소스
C
#include "mfrc522_picc.h"
// NXP MFRC522 Rev 3.9 Register Addresses.
enum {
    CommandReg=0x01, ComIrqReg=0x04, ErrorReg=0x06,
    FIFODataReg=0x09, FIFOLevelReg=0x0A,
    ControlReg=0x0C, BitFramingReg=0x0D
};
// PCD_Idle=0x00, PCD_Transceive=0x0C; StartSend bit7.
static rc522_status_t write_reg(const rc522_bus_t *bus,uint8_t reg,uint8_t val)
{
    return bus->write(bus->ctx,reg,val)==0 ? RC522_OK : RC522_IO_ERROR;
}
static rc522_status_t read_reg(const rc522_bus_t *bus,uint8_t reg,uint8_t *val)
{
    return bus->read(bus->ctx,reg,val)==0 ? RC522_OK : RC522_IO_ERROR;
}

static rc522_status_t transceive(const rc522_bus_t *bus,
    const uint8_t *tx,size_t tx_len,uint8_t tx_last_bits,
    uint8_t *rx,size_t expected_len,uint32_t max_polls)
{
    uint8_t irq,err,level,last_bits,byte;
    size_t i;
    uint32_t tries;
    rc522_status_t rc;
    if(tx_len==0 || tx_len>64 || expected_len==0 || expected_len>64 ||
       max_polls==0 || tx_last_bits>7) return RC522_PROTOCOL_ERROR;
#define WR(REG,VALUE) do { rc=write_reg(bus,(REG),(VALUE)); if(rc) return rc; } while(0)
#define RD(REG,VALUE) do { rc=read_reg(bus,(REG),(VALUE)); if(rc) return rc; } while(0)
    WR(CommandReg,0x00);       // Idle
    WR(ComIrqReg,0x7F);        // Clear pending IRQ
    WR(FIFOLevelReg,0x80);     // Flush FIFO
    for(i=0;i<tx_len;i++) WR(FIFODataReg,tx[i]);
    WR(BitFramingReg,tx_last_bits); // BitFramingReg.TxLastBits
    WR(CommandReg,0x0C);      // Transceive
    WR(BitFramingReg,(uint8_t)(0x80u|tx_last_bits)); // StartSend

    irq=0;
    for(tries=0;tries<max_polls;tries++) {
        RD(ComIrqReg,&irq);
        if(irq & 0x30) break;    // RxIrq / IdleIrq
        if(irq & 0x01) break;    // TimerIrq
    }
    // Clear StartSend after device completion or timeout.
    WR(BitFramingReg,tx_last_bits);
    if(tries==max_polls || (irq & 0x01)) return RC522_TIMEOUT;
    RD(ErrorReg,&err);
    if(err & 0x08) return RC522_COLLISION;
    if(err & 0x13) return RC522_PROTOCOL_ERROR;
    RD(FIFOLevelReg,&level);
    RD(ControlReg,&last_bits);
    if(level!=expected_len || (last_bits & 0x07)!=0)
        return RC522_PROTOCOL_ERROR;
    for(i=0;i<expected_len;i++) {
        RD(FIFODataReg,&byte);
        rx[i]=byte;
    }
#undef WR
#undef RD
    return RC522_OK;
}
rc522_status_t rc522_read_uid4(const rc522_bus_t *bus,uint8_t uid[4],
                                uint8_t atqa[2],uint32_t max_polls)
{
    const uint8_t reqa=0x26;
    const uint8_t anticoll[2]={0x93,0x20};
    uint8_t answer[5];
    uint8_t bcc;
    rc522_status_t result;
    if(!bus || !bus->read || !bus->write || !uid || !atqa)
        return RC522_PROTOCOL_ERROR;
    result=transceive(bus,&reqa,1,7,atqa,2,max_polls);
    if(result!=RC522_OK) return result;
    result=transceive(bus,anticoll,2,0,answer,5,max_polls);
    if(result!=RC522_OK) return result;
    // 0x88 Cascade Tag: UID is 7/10 bytes; not supported in this lesson.
    if(answer[0]==0x88) return RC522_UNSUPPORTED_UID;
    bcc=(uint8_t)(answer[0]^answer[1]^answer[2]^answer[3]);
    if(bcc!=answer[4]) return RC522_PROTOCOL_ERROR;
    for(size_t i=0;i<4;i++) uid[i]=answer[i];
    return RC522_OK;
}

                  m->fifo_tx[1]==0x20) {

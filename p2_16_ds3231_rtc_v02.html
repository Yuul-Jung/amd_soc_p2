01과제 개요

DS3231(디에스3231)은 I²C(아이투씨, 두 선식 직렬 통신) 기반의 RTC(Real-Time Clock, 실시간 시계) 장치다. 시간·날짜를 내장 발진기로 유지하고 백업 배터리 전원도 지원한다. DS18B20의 Dallas 1-Wire와는 프로토콜이 다르다.

이번 실습은 DS3231의 시간·날짜 7바이트 레지스터 읽기 순서를 설계하고, BCD(이진화 십진수)를 십진수로 변환하여 정상 시각과 오류를 구분한다. 기존 제14장 I²C 단일 바이트 쓰기 수업에서 Repeated START(반복 시작)와 연속 읽기로 확장되는 부분을 학습한다.

02계층형 시스템 구조

현재 경로는 Sequencer + 모의 버스 응답 및 C + 모의 DS3231 레지스터 두 개다. 가운데의 실제 I²C Read Master는 아직 연결되지 않았으므로 임의로 실장 성공으로 표시하지 않는다.

03I²C 장치 주소와 레지스터

DS3231의 7비트 장치 주소는 0x68이다. I²C 버스에서 8비트 주소 프레임으로 송신할 때 쓰기는 0xD0, 읽기는 0xD1이다.

레지스터 주소내용표현/주의
0x00Seconds(초)BCD 00~59
0x01Minutes(분)BCD 00~59
0x02Hours(시)12/24시간제 선택 및 AM/PM
0x03Day(요일 번호)1~7, 요일 이름과 숫자 연결은 사용자 설정
0x04Date(일)01~31, 실제 월별 유효 범위 검사
0x05Month/Century1~12, bit7 Century
0x06Year(년)00~99, 교육용 2000년 기준 변환
0x0FStatus(상태)bit7 OSF: 시계 발진 중단 이력

OSF(Oscillator Stop Flag, 발진 중단 표시)가 1이라면 읽힌 날짜·시간을 곧바로 신뢰하지 않는다. 초기 시각 설정과 플래그 정리가 필요한지는 실제 장치 상태에 따라 결정한다. 본 실습에서 OSF를 자동으로 지우지 않는다. Analog Devices 공식 DS3231 데이터시트.

04Repeated START를 사용하는 7바이트 읽기

실습 명령 · Bash / 코드 실행 예시
명령·출력
START
  → 0xD0 : DS3231 쓰기 주소
  → ACK
  → 0x00 : 시작 레지스터 포인터(Seconds)
  → ACK
  → REPEATED START
  → 0xD1 : DS3231 읽기 주소
  → ACK
  → Read Seconds(0x00) → ACK
  → Read Minutes(0x01) → ACK
  → Read Hours(0x02)   → ACK
  → Read Weekday(0x03) → ACK
  → Read Date(0x04)    → ACK
  → Read Month(0x05)   → ACK
  → Read Year(0x06)    → NACK
  → STOP

마지막 바이트 후 마스터는 NACK을 보내고 STOP으로 종료한다. 원래 14장 i2c_write_byte_master는 쓰기 전용이므로 그대로 16장 읽기를 지원한다고 보고할 수 없다. 16장의 ds3231_burst_sequencer는 하위 I²C 버스 엔진에 전달할 명령만 만든다.

05BCD 변환과 날짜의 유효성

BCD는 10진수 각 자리(0~9)를 4비트씩 인코딩하는 표현이다. 예를 들어 0x42는 16진 정수 66이 아니라 42분이다.

Raw 데이터변환결과
Seconds 0x585×10 + 858초
Minutes 0x424×10 + 242분
Hours 0x090×10 + 909시(24시간제)
Date 0x111×10 + 111일
Month 0x111×10 + 111월
Year 0x2620 + 62026년(2000 기준)

12시간제에서는 12 AM → 00시, 12 PM → 12시, 1 PM → 13시로 변환한다. 실제 날짜 확인에서 윤년·월별 일수도 확인한다. 예를 들어 2024년 2월 29일은 유효하지만 2100년 2월 29일은 유효하지 않다.

DS3231 내부 연도를 무조건 컴퓨터 현재 날짜로 간주하지 않는다. BCD 값 자체가 올바르더라도 OSF 및 초기설정 상태를 확인하지 않은 경우 시각의 실제 정확도는 미확인이다.

#include "ds3231_time.h"
static int bcd_to_uint(uint8_t v,int *out)
{
    if ((v & 0x0fu)>9u || ((v>>4)&0x0fu)>9u) return -1;
    *out=(int)((v>>4)*10u+(v&0x0fu));
    return 0;
}
static int leap(unsigned y)
{
    return (y%4u==0u && (y%100u!=0u || y%400u==0u));
}
ds3231_result_t ds3231_decode_datetime(const uint8_t raw[7],
                                        ds3231_datetime_t *out)
{
    ds3231_datetime_t t;
    int seconds,minutes,hours,date,month,year,weekday,dim;
    const int days[12]={31,28,31,30,31,30,31,31,30,31,30,31};
    if(!raw||!out) return DS3231_ERR_ARG;
    if((raw[0]&0x80u) || (raw[1]&0x80u) || (raw[2]&0x80u) ||
       (raw[3]&0xf8u) || (raw[4]&0xc0u) || (raw[5]&0x60u))
        return DS3231_ERR_RANGE;
    if(bcd_to_uint(raw[0]&0x7fu,&seconds) ||
       bcd_to_uint(raw[1]&0x7fu,&minutes) ||
       bcd_to_uint(raw[3]&0x07u,&weekday) ||
       bcd_to_uint(raw[4]&0x3fu,&date) ||
       bcd_to_uint(raw[5]&0x1fu,&month) ||
       bcd_to_uint(raw[6],&year)) return DS3231_ERR_BCD;
    if(raw[2]&0x40u) {
        // DS3231 hours register bit6=1: 12-hour, bit5=PM.
        int h12;
        if((raw[2]&0x80u) || bcd_to_uint(raw[2]&0x1fu,&h12))
            return DS3231_ERR_BCD;
        if(h12<1 || h12>12) return DS3231_ERR_RANGE;
        hours=h12%12 + ((raw[2]&0x20u)?12:0);
    } else {
        if(bcd_to_uint(raw[2]&0x3fu,&hours)) return DS3231_ERR_BCD;
    }
    t.year=(uint16_t)(2000+year+((raw[5]&0x80u)?100:0));
    if(seconds>59 || minutes>59 || hours>23 || weekday<1 ||
       weekday>7 || month<1 || month>12 || date<1)
        return DS3231_ERR_RANGE;
    dim=days[month-1];
    if(month==2 && leap(t.year)) dim++;
    if(date>dim) return DS3231_ERR_RANGE;
    t.second=(uint8_t)seconds;t.minute=(uint8_t)minutes;
    t.hour=(uint8_t)hours;t.day=(uint8_t)date;
    t.month=(uint8_t)month;t.weekday=(uint8_t)weekday;
    *out=t; return DS3231_OK;
}
ds3231_result_t ds3231_read_datetime(ds3231_read_regs_fn reader, void *ctx,
                                      ds3231_datetime_t *out)
{
    uint8_t raw[7],status;
    if(!reader||!out) return DS3231_ERR_ARG;
    // Check OSF before and after snapshot to reject oscillator-stop
    // conditions arising during the read.
    if(reader(ctx,DS3231_I2C_ADDR,0x0f,&status,1)!=0)
        return DS3231_ERR_BUS;
    if(status&0x80u) return DS3231_ERR_OSF;
    if(reader(ctx,DS3231_I2C_ADDR,0x00,raw,7)!=0)
        return DS3231_ERR_BUS;
    if(reader(ctx,DS3231_I2C_ADDR,0x0f,&status,1)!=0)
        return DS3231_ERR_BUS;
    if(status&0x80u) return DS3231_ERR_OSF;
    return ds3231_decode_datetime(raw,out);
}

`timescale 1ns/1ps
// P2-16 transaction-level RTC register read planner (NOT physical I2C).
// Valid/ready command interface to a FUTURE I2C write/read engine.
// Sequence: START, D0(write address), 00(pointer), RESTART,
//           D1(read address), 6 READ_ACK, READ_NACK, STOP.
// Each bus command produces one response pulse; READ responses carry data.
// I2C opcode: 0 START,1 WRITE_BYTE,2 RESTART,3 READ_ACK,
//             4 READ_NACK,5 STOP. No electrical SCL/SDA RTL in this module.
module ds3231_burst_sequencer (
    input wire clk,input wire resetn,input wire start_pulse,
    output reg busy,output reg done_pulse,output reg error_pulse,
    output wire bus_valid,input wire bus_ready,
    output reg [2:0] bus_op,output reg [7:0] bus_wdata,
    input wire bus_rsp_valid,input wire bus_rsp_error,
    input wire [7:0] bus_rdata,
    output reg [55:0] raw_bcd // byte0 seconds is raw_bcd[7:0]
);
    localparam [1:0] IDLE=0, SEND=1, WAIT_RSP=2, FINISH=3;
    reg [1:0] state;
    reg [3:0] step;
    reg [55:0] shifted;
    reg [7:0] tmp_data;
    assign bus_valid=(state==SEND);
    always @(*) begin
        bus_op=0;bus_wdata=0;
        case(step)
        0:bus_op=0;                       // START
        1:begin bus_op=1;bus_wdata=8'hd0;end
        2:begin bus_op=1;bus_wdata=8'h00;end
        3:bus_op=2;                       // REPEATED START
        4:begin bus_op=1;bus_wdata=8'hd1;end
        5,6,7,8,9,10:bus_op=3;           // read + ACK
        11:bus_op=4;                      // read + NACK final
        12:bus_op=5;                      // STOP
        default:bus_op=0;
        endcase
    end
    always @(posedge clk) begin
        if(!resetn) begin
            state<=IDLE;step<=0;busy<=0;done_pulse<=0;
            error_pulse<=0;raw_bcd<=0;shifted<=0;tmp_data<=0;
        end else begin
            done_pulse<=0;error_pulse<=0;
            case(state)
            IDLE:if(start_pulse)begin
                step<=0;busy<=1;shifted<=0;state<=SEND;
            end
            SEND:if(bus_ready)state<=WAIT_RSP;
            WAIT_RSP:if(bus_rsp_valid)begin
                if(bus_rsp_error)begin
                    // Report failure; do not publish partial data.
                    // Actual I2C engine shall release bus on error.
                    error_pulse<=1;busy<=0;state<=IDLE;
                end else begin
                    if(step>=5 && step<=11)
                        shifted[8*(step-5)+:8]<=bus_rdata;
                    if(step==12)begin
                        raw_bcd<=shifted;state<=FINISH;
                    end else begin step<=step+1'b1;state<=SEND;end
                end
            end
            FINISH:begin
                done_pulse<=1;busy<=0;state<=IDLE;
            end
            default:state<=IDLE;
            endcase
        end
    end
endmodule

#include <assert.h>
#include <stdio.h>
#include <string.h>
#include "../sw/ds3231_time.h"
typedef struct {uint8_t regs[0x13];int calls,fail_at,osf_during_read;} fake_rtc_t;
static int fake_read(void *ctx,uint8_t addr,uint8_t reg,uint8_t *dest,size_t len)
{
    fake_rtc_t *f=(fake_rtc_t*)ctx;
    f->calls++;
    if(f->fail_at==f->calls || addr!=0x68 || (size_t)reg+len>sizeof(f->regs))
        return -1;
    if(f->osf_during_read && f->calls==3) f->regs[0x0f]|=0x80;
    memcpy(dest,f->regs+reg,len);
    return 0;
}
static void seed(fake_rtc_t *f)
{
    memset(f,0,sizeof(*f));
    // 2026-11-11 09:42:58, user-defined weekday 4.
    uint8_t v[7]={0x58,0x42,0x09,0x04,0x11,0x11,0x26};
    memcpy(f->regs,v,7);
}
int main(void)
{
    fake_rtc_t f;ds3231_datetime_t t;
    seed(&f);
    assert(ds3231_read_datetime(fake_read,&f,&t)==DS3231_OK);
    assert(f.calls==3 && t.year==2026 && t.month==11 && t.day==11 &&
           t.hour==9 && t.minute==42 && t.second==58 && t.weekday==4);
    puts("P2_16_RTC_BURST_BCD_24H_PASS");
    seed(&f);f.regs[2]=0x72; // 12-hour 12 PM -> 12:xx
    assert(ds3231_read_datetime(fake_read,&f,&t)==DS3231_OK && t.hour==12);
    f.calls=0;f.regs[2]=0x52; // 12 AM -> 00:xx
    assert(ds3231_read_datetime(fake_read,&f,&t)==DS3231_OK && t.hour==0);
    f.calls=0;f.regs[2]=0x61; // 1 PM -> 13:xx
    assert(ds3231_read_datetime(fake_read,&f,&t)==DS3231_OK && t.hour==13);
    puts("P2_16_RTC_12H_AM_PM_PASS");
    seed(&f);f.regs[4]=0x29;f.regs[5]=0x02;f.regs[6]=0x24;
    assert(ds3231_decode_datetime(f.regs,&t)==DS3231_OK && t.day==29);
    f.regs[6]=0x25;
    assert(ds3231_decode_datetime(f.regs,&t)==DS3231_ERR_RANGE);
    f.regs[4]=0x31;f.regs[5]=0x04;
    assert(ds3231_decode_datetime(f.regs,&t)==DS3231_ERR_RANGE);
    puts("P2_16_RTC_LEAP_CALENDAR_PASS");
    seed(&f);f.regs[5]=0x91;f.regs[6]=0x00;f.regs[4]=0x11;
    assert(ds3231_decode_datetime(f.regs,&t)==DS3231_OK && t.year==2100);
    f.regs[4]=0x29;f.regs[5]=0x82;
    assert(ds3231_decode_datetime(f.regs,&t)==DS3231_ERR_RANGE);
    puts("P2_16_RTC_CENTURY_2100_PASS");
    seed(&f);f.regs[1]=0x6a;
    assert(ds3231_decode_datetime(f.regs,&t)==DS3231_ERR_BCD);
    seed(&f);f.regs[0]=0x60;
    assert(ds3231_decode_datetime(f.regs,&t)==DS3231_ERR_RANGE);
    seed(&f);f.regs[3]=0;
    assert(ds3231_decode_datetime(f.regs,&t)==DS3231_ERR_RANGE);
    puts("P2_16_RTC_INVALID_DATA_REJECT_PASS");
    seed(&f);f.regs[0x0f]=0x80;
    assert(ds3231_read_datetime(fake_read,&f,&t)==DS3231_ERR_OSF);
    seed(&f);f.osf_during_read=1;
    assert(ds3231_read_datetime(fake_read,&f,&t)==DS3231_ERR_OSF);
    seed(&f);f.fail_at=2;
    assert(ds3231_read_datetime(fake_read,&f,&t)==DS3231_ERR_BUS);
    puts("P2_16_RTC_OSF_BUS_ERROR_PASS");
    puts("P2_16_HOST_RTC_TESTS_PASS");
    return 0;
}

// Fake transaction responder verifies exact DS3231 repeated-start
// sequence, reads 7 BCD register bytes, and injects a bus error.
module tb_ds3231_burst_sequencer;
    reg clk=0;always #5 clk=~clk;
    reg resetn=0,start=0,ready=1;
    wire busy,done,error,valid;
    wire [2:0] op;
    wire [7:0] wdata;
    wire [55:0] raw;
    reg rsp_valid=0,rsp_error=0;
    reg [7:0] rsp_data=0;
    reg [7:0] sample [0:6];
    integer cmd_count=0,read_count=0,mode=0;
    ds3231_burst_sequencer dut(
        .clk(clk),.resetn(resetn),.start_pulse(start),
        .busy(busy),.done_pulse(done),.error_pulse(error),
        .bus_valid(valid),.bus_ready(ready),
        .bus_op(op),.bus_wdata(wdata),
        .bus_rsp_valid(rsp_valid),.bus_rsp_error(rsp_error),
        .bus_rdata(rsp_data),.raw_bcd(raw)
    );
    reg queued=0, queued_error=0;
    reg [7:0] queued_data=0;
    // (변경) 유효한 bus_valid/bus_ready 핸드셰이크는 상승 에지에서 수락.
    // 다음 반주기에서 응답을 준비하여 WAIT_RSP 상승 에지까지 유지한다.
    always @(posedge clk) begin
        if(!resetn) queued<=0;
        else begin
            queued<=0;
            if(valid && ready)begin
                if(cmd_count>12) $fatal(1,"too many commands");
                case(cmd_count)
                0:if(op!==0) $fatal(1,"expected START");
                1:if(op!==1 || wdata!==8'hd0) $fatal(1,"expected D0");
                2:if(op!==1 || wdata!==8'h00) $fatal(1,"expected pointer00");
                3:if(op!==2) $fatal(1,"expected RESTART");
                4:if(op!==1 || wdata!==8'hd1) $fatal(1,"expected D1");
                5,6,7,8,9,10:if(op!==3) $fatal(1,"expected READ_ACK");
                11:if(op!==4) $fatal(1,"expected READ_NACK");
                12:if(op!==5) $fatal(1,"expected STOP");
                endcase
                if(cmd_count>=5 && cmd_count<=11)begin
                    queued_data=sample[read_count];
                    read_count=read_count+1;
                end else queued_data=0;
                queued_error=(mode==1 && cmd_count==4);
                cmd_count=cmd_count+1;
                queued<=1;
            end
        end
    end
    always @(negedge clk) begin
        rsp_valid=queued;
        rsp_error=queued_error;
        rsp_data=queued_data;
    end
    task begin_test;
        input integer test_mode;
        begin
            @(negedge clk);mode=test_mode;
            cmd_count=0;read_count=0;start=1;
            @(negedge clk);start=0;
        end
    endtask
    initial begin
        sample[0]=8'h58;sample[1]=8'h42;sample[2]=8'h09;
        sample[3]=8'h04;sample[4]=8'h11;sample[5]=8'h11;
        sample[6]=8'h26;
        repeat(3) @(negedge clk);resetn=1;
        begin_test(0);
        wait(done);#1;
        if(raw!==56'h26111104094258 ||
           cmd_count!=13 || read_count!=7 || busy)
            $fatal(1,"DS3231 burst or packing error %h cmds=%0d reads=%0d",
                   raw,cmd_count,read_count);
        $display("P2_16_RTC_I2C_BURST_SEQUENCE_PASS");
        begin_test(1);
        wait(error);#1;

07전기적 연결과 후속 확장

DS3231의 SDA·SCL은 오픈드레인 I²C 신호이며 3.3V 풀업을 사용해야 한다. 흔한 DS3231 모듈 중에는 5V 전원·충전 배터리 회로·5V 풀업을 포함하는 제품도 있으므로 현물 회로와 풀업 전압을 검토하기 전 Basys3와 직접 연결하지 않는다. 실제 XDC는 보드의 기존 핀 배정을 확인한 다음 정한다.

16.1 BCD 시간 읽기 → 16.2 7바이트 버스트 읽기 → 16.3 상태/OSF 확인 → 16.4 완전한 I²C Read Master 및 LCD/FND 시간 표시로 확장한다. 이미 작성된 RTL Sequencer와 C 확인 코드를 재사용한다.
